-- 1) Tables (only created when missing - your existing data is untouched)
create table if not exists public.rings (
  id int generated by default as identity primary key,
  ring_name text not null default 'Ring 1',
  ring_code text,
  laptop_password text,
  phone_password text,
  tv_code text,
  max_phones int not null default 4
);

create table if not exists public.matches (
  id int generated by default as identity primary key,
  ring_id int not null unique,
  blue_score int not null default 0,
  red_score int not null default 0,
  blue_gam_jeom int not null default 0,
  red_gam_jeom int not null default 0,
  blue_yellow_cards int not null default 0,
  red_yellow_cards int not null default 0,
  judge_blue int[] not null default '{0,0,0,0}',
  judge_red int[] not null default '{0,0,0,0}',
  updated_at timestamptz not null default now()
);

create table if not exists public.phones (
  id text primary key,
  ring_id int not null,
  judge_slot int not null,
  last_seen timestamptz not null default now()
);

do $$
declare fn record;
begin
  for fn in
    select p.oid::regprocedure::text as sig
    from pg_proc p
    join pg_namespace n on n.oid = p.pronamespace
    where n.nspname = 'public'
      and p.proname in ('judge_score','adjust_judge_score','adjust_card',

-- 6) Helper
create function public.norm_judge_array(a int[])
returns int[] language sql immutable as
$$ select array[coalesce(a[1],0), coalesce(a[2],0), coalesce(a[3],0), coalesce(a[4],0)] $$;

-- 7) Phone list - NO heartbeat-age filter = NO automatic kicking.
--    A phone only disappears when someone presses KICK (or it logs out).
create function public.get_active_phones(p_ring_id int)
returns table (id text, judge_slot int, last_seen timestamptz)
language sql stable security definer as
$$
  select ph.id, ph.judge_slot, ph.last_seen
  from public.phones ph
  where ph.ring_id = p_ring_id
  order by ph.judge_slot;
$$;

-- 8) Register a judge phone - enforces the referee limit (3 or 4).
--    When the party is set to 3, a 4th phone is rejected here.
create function public.phone_register(p_id text, p_ring_id int, p_slot int)
returns void language plpgsql security definer as
$$
declare v_max int;
begin
  select coalesce(r.max_phones, 4) into v_max from public.rings r where r.id = p_ring_id;
  if not found then raise exception 'Ring not found'; end if;
  if v_max <> 3 then v_max := 4; end if;

  if p_slot < 1 or p_slot > v_max then
    raise exception 'This ring only allows % referees', v_max;
  end if;

  if exists (select 1 from public.phones
             where ring_id = p_ring_id and judge_slot = p_slot and id <> p_id) then
    raise exception 'Corner % is already taken', p_slot;
  end if;

  if (select count(*) from public.phones
      where ring_id = p_ring_id and id <> p_id) >= v_max then
    raise exception 'This ring is full - max % referees', v_max;
  end if;

  insert into public.phones (id, ring_id, judge_slot, last_seen)
  values (p_id, p_ring_id, p_slot, now())
  on conflict (id) do update
  set ring_id = excluded.ring_id, judge_slot = excluded.judge_slot, last_seen = now();
end $$;

create function public.phone_unregister(p_id text)
returns void language sql security definer as
$$ delete from public.phones where id = p_id $$;

create function public.phone_heartbeat(p_id text)
returns void language plpgsql security definer as
$$
begin
  update public.phones set last_seen = now() where id = p_id;
  if not found then raise exception 'Phone not registered'; end if;
end $$;

-- 9) Judge scoring - adds points to the judge corner AND to the big match
--    score. Big scores are cumulative now, never capped at 0-4.
create function public.judge_score(p_ring_id int, p_slot int, p_side text, p_points int, p_session text)
returns void language plpgsql security definer as
$$
declare v_jb int[]; v_jr int[];
begin
  if p_side not in ('blue','red') then raise exception 'Invalid side'; end if;
  if p_points < 1 or p_points > 5 then raise exception 'Points must be 1-5'; end if;
  if p_slot < 1 or p_slot > 4 then raise exception 'Invalid judge slot'; end if;

  if not exists (select 1 from public.phones
                 where id = p_session and ring_id = p_ring_id and judge_slot = p_slot) then
    raise exception 'Phone not registered - please log in again';
  end if;

  update public.phones set last_seen = now() where id = p_session;

  insert into public.matches (ring_id)
  select p_ring_id
  where not exists (select 1 from public.matches where ring_id = p_ring_id);

  select public.norm_judge_array(judge_blue), public.norm_judge_array(judge_red)
  into v_jb, v_jr
  from public.matches where ring_id = p_ring_id;

  if p_side = 'blue' then v_jb[p_slot] := v_jb[p_slot] + p_points;
  else v_jr[p_slot] := v_jr[p_slot] + p_points; end if;

  update public.matches set
    judge_blue = v_jb,
    judge_red  = v_jr,
    blue_score = greatest(0, blue_score + case when p_side = 'blue' then p_points else 0 end),
    red_score  = greatest(0, red_score  + case when p_side = 'red'  then p_points else 0 end),
    updated_at = now()
  where ring_id = p_ring_id;
end $$;

-- 10) Laptop +/- buttons on judge corners
create function public.adjust_judge_score(p_ring_id int, p_slot int, p_side text, p_delta int)
returns void language plpgsql security definer as
$$
declare v_jb int[]; v_jr int[]; v_old int; v_new int; v_applied int;
begin
  if p_side not in ('blue','red') then raise exception 'Invalid side'; end if;
  if p_slot < 1 or p_slot > 4 then raise exception 'Invalid judge slot'; end if;

  insert into public.matches (ring_id)
  select p_ring_id
  where not exists (select 1 from public.matches where ring_id = p_ring_id);

  select public.norm_judge_array(judge_blue), public.norm_judge_array(judge_red)
  into v_jb, v_jr
  from public.matches where ring_id = p_ring_id;

  if p_side = 'blue' then
    v_old := v_jb[p_slot]; v_new := greatest(0, v_old + p_delta); v_jb[p_slot] := v_new;
  else
    v_old := v_jr[p_slot]; v_new := greatest(0, v_old + p_delta); v_jr[p_slot] := v_new;
  end if;
  v_applied := v_new - v_old;

  update public.matches set
    judge_blue = v_jb,
    judge_red  = v_jr,
    blue_score = greatest(0, blue_score + case when p_side = 'blue' then v_applied else 0 end),
    red_score  = greatest(0, red_score  + case when p_side = 'red'  then v_applied else 0 end),
    updated_at = now()
  where ring_id = p_ring_id;
end $$;

-- 11) Laptop +/- buttons on score / gam-jeom / yellow cards
create function public.adjust_card(p_ring_id int, p_column text, p_delta int)
returns void language plpgsql security definer as
$$
begin
  if p_column not in ('blue_score','red_score','blue_gam_jeom','red_gam_jeom',
                      'blue_yellow_cards','red_yellow_cards') then
    raise exception 'Invalid column';
  end if;

  insert into public.matches (ring_id)
  select p_ring_id
  where not exists (select 1 from public.matches where ring_id = p_ring_id);

  execute format(
    'update public.matches set %I = greatest(0, %I + $1), updated_at = now() where ring_id = $2',
    p_column, p_column
  ) using p_delta, p_ring_id;
end $$;
